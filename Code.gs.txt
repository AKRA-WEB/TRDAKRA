function getProductUnit(itemName) {
  try {
    var sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(PRODUCT_SHEET_NAME);
    if (!sheet) return "หน่วย";
    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    if (lastRow < 2) return "หน่วย";
    var rangeValues = sheet.getRange(1, 1, lastRow, lastCol).getValues();
    var headers = rangeValues[0];
    var nameColIndex = headers.indexOf("Product name");
    if (nameColIndex === -1) nameColIndex = 1;
    var unitColIndex = headers.indexOf("Unit");
    if (unitColIndex === -1) unitColIndex = headers.indexOf("unit");
    if (unitColIndex === -1) unitColIndex = headers.indexOf("หน่วย");
    if (unitColIndex === -1) { unitColIndex = nameColIndex + 1; if (unitColIndex >= lastCol) unitColIndex = lastCol - 1; }
    for (var i = 1; i < rangeValues.length; i++) {
      var name = rangeValues[i][nameColIndex];
      if (name && name.toString().trim() === itemName.toString().trim()) {
        var unitValue = rangeValues[i][unitColIndex];
        if (unitValue) return unitValue.toString().trim();
      }
    }
  } catch (error) { Logger.log("Error in getProductUnit: " + error.toString()); }
  return "หน่วย";
}

function doGet(e) {
  if (!e || !e.parameter) {
    return ContentService.createTextOutput("API is running.");
  }
  var action = e.parameter.action;

  if (action === 'getProducts') {
    return ContentService.createTextOutput(JSON.stringify(getProductList()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (action === 'getItems') {
    return ContentService.createTextOutput(JSON.stringify(getInventoryData()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (action === 'getSurveyLog') {
    return ContentService.createTextOutput(JSON.stringify(getSurveyLog()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var defaultData = getInventoryData();
  return ContentService.createTextOutput(JSON.stringify(defaultData))
    .setMimeType(ContentService.MimeType.JSON);
}

// 4. ฟังก์ชันดึง "รายชื่อสินค้า" จากชีต Product
//    คืนค่า: name, unit, floor, location, parLevel
// ===============================================
function getProductList() {
  try {
    var sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(PRODUCT_SHEET_NAME);
    if (!sheet) return [];
    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    if (lastRow < 2) return [];

    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    // หาตำแหน่งคอลัมน์ที่มีอยู่
    var nameColIndex = headers.indexOf("Product name");
    if (nameColIndex === -1) nameColIndex = 1;
    var unitColIndex = headers.indexOf("Unit");
    if (unitColIndex === -1) unitColIndex = headers.indexOf("unit");
    if (unitColIndex === -1) unitColIndex = headers.indexOf("หน่วย");
    if (unitColIndex === -1) { unitColIndex = nameColIndex + 1; if (unitColIndex >= lastCol) unitColIndex = lastCol - 1; }

    // คอลัมน์ใหม่ (ถ้าไม่มีใน Sheet จะคืนค่าว่าง)
    var floorColIndex    = headers.indexOf("Floor");
    var locationColIndex = headers.indexOf("Location");
    var parLevelColIndex = headers.indexOf("Par Level");

    var dataValues = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
    var products = [];

    for (var i = 0; i < dataValues.length; i++) {
      var name = dataValues[i][nameColIndex];
      if (name) {
        products.push({
          name:     name.toString().trim(),
          unit:     unitColIndex >= 0 && dataValues[i][unitColIndex] ? dataValues[i][unitColIndex].toString().trim() : "หน่วย",
          floor:    floorColIndex >= 0    ? (dataValues[i][floorColIndex]    || "").toString().trim() : "",
          location: locationColIndex >= 0 ? (dataValues[i][locationColIndex] || "").toString().trim() : "",
          parLevel: parLevelColIndex >= 0 ? (parseInt(dataValues[i][parLevelColIndex]) || 0) : 0
        });
      }
    }
    return products;
  } catch (error) {
    Logger.log("Error in getProductList: " + error.toString());
    return [];
  }
}

// ===============================================
// 5. ฟังก์ชัน (ใหม่) อัปเดต Floor/Location/ParLevel
//    กลับลงชีต Product ทีละรายการสินค้า
// ===============================================
function updateProductDetails(itemName, floor, location, parLevel) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName(PRODUCT_SHEET_NAME);
    if (!sheet) return false;

    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    if (lastRow < 2) return false;

    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var nameColIndex = headers.indexOf("Product name");
    if (nameColIndex === -1) nameColIndex = 1;

    // สร้างคอลัมน์ใหม่ถ้ายังไม่มี
    var floorColIndex    = headers.indexOf("Floor");
    var locationColIndex = headers.indexOf("Location");
    var parLevelColIndex = headers.indexOf("Par Level");

    if (floorColIndex === -1) {
      floorColIndex = lastCol;
      sheet.getRange(1, lastCol + 1).setValue("Floor");
      lastCol++;
    }
    if (locationColIndex === -1) {
      locationColIndex = lastCol;
      sheet.getRange(1, lastCol + 1).setValue("Location");
      lastCol++;
    }
    if (parLevelColIndex === -1) {
      parLevelColIndex = lastCol;
      sheet.getRange(1, lastCol + 1).setValue("Par Level");
      lastCol++;
    }

    // ค้นหาแถวสินค้าแล้วอัปเดต
    var nameValues = sheet.getRange(2, nameColIndex + 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < nameValues.length; i++) {
      if (nameValues[i][0] && nameValues[i][0].toString().trim() === itemName.toString().trim()) {
        var rowNum = i + 2; // +2 เพราะ header อยู่แถว 1
        if (floor    !== undefined && floor    !== null) sheet.getRange(rowNum, floorColIndex + 1).setValue(floor);
        if (location !== undefined && location !== null) sheet.getRange(rowNum, locationColIndex + 1).setValue(location);
        if (parLevel !== undefined && parLevel !== null && parLevel > 0) sheet.getRange(rowNum, parLevelColIndex + 1).setValue(parseInt(parLevel));
        return true;
      }
    }
    return false; // ไม่พบสินค้า
  } catch (error) {
    Logger.log("Error in updateProductDetails: " + error.toString());
    return false;
  }
}

function getInventoryData() {

    for (var i = 1; i < data.length; i++) {
      if (!data[i][0]) continue;
      result.push({
        id: data[i][0],
        timestamp: data[i][1],
        itemName: data[i][2],
        requestQty: data[i][3],
        status: data[i][4],
        oldExpiry: data[i][5] || 'ไม่ได้ระบุ',
        newExpiry: data[i][6] || '',
        receiveQty: data[i][7] === '' ? null : data[i][7],
        receiveNote: data[i][8] || '',
        w2Note: data[i][9] || '',
        storageCapacity: data[i][10] || 0,
        dispatchTimestamp: data[i][11] || '',
        requestedBy: data[i][12] || ''
      });
    }
    return result.reverse();
  } catch (error) { return []; }
}

function updateInventoryData(items) {
  try {
    var sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME);
    if (!sheet) return false;
    if (sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).clearContent();
    }
    if (items && items.length > 0) {
      var itemsToSave = items.slice().reverse();
      var rows = itemsToSave.map(function(item) {
        return [
          item.id, item.timestamp, item.itemName, item.requestQty,
          item.status, item.oldExpiry, item.newExpiry, item.receiveQty,
          item.receiveNote, item.w2Note || '', item.storageCapacity || 0,
          item.dispatchTimestamp || '', item.requestedBy || ''
        ];
      });
      sheet.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
    }
    return true;
  } catch (error) { return false; }
}

// 9a. สรุปการจัดส่งสินค้าประจำวัน (Dispatch) → LINE
//     ตั้ง Time-based trigger ให้รันทุกวัน 17:00–18:00 Bangkok
//     (= 10:00–11:00 UTC เนื่องจาก Bangkok = UTC+7)
// ===============================================
function sendDailyDispatchSummary() {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet || sheet.getLastRow() < 2) {
      Logger.log("sendDailyDispatchSummary: no data rows");
      return;
    }

    var now     = new Date();
    var todayDD = Utilities.formatDate(now, "Asia/Bangkok", "dd");
    var todayMM = Utilities.formatDate(now, "Asia/Bangkok", "MM");
    var prefix  = todayDD + "-" + todayMM;

    // Columns (0-based): id(0), timestamp(1), itemName(2), requestQty(3),
    //   status(4), oldExpiry(5), newExpiry(6), receiveQty(7), receiveNote(8),
    //   w2Note(9), storageCapacity(10), dispatchTimestamp(11), requestedBy(12)
    var lastRow = sheet.getLastRow();
    var numCols = Math.min(sheet.getLastColumn(), 13);
    var data    = sheet.getRange(2, 1, lastRow - 1, numCols).getValues();

    var dispatched = [], outstock = [], pending = [];

    for (var i = 0; i < data.length; i++) {
      var ts       = data[i][1] ? data[i][1].toString() : '';
      var itemName = data[i][2] ? data[i][2].toString().trim() : '';
      var reqQty   = data[i][3] ? data[i][3].toString() : '';
      var status   = data[i][4] ? data[i][4].toString().trim() : '';

      if (!itemName || !ts.startsWith(prefix)) continue;

      if (status === 'จัดส่งแล้ว') {
        dispatched.push({ name: itemName, qty: reqQty });
      } else if (status === 'สินค้าหมด') {
        outstock.push({ name: itemName, qty: reqQty });
      } else if (status === 'สั่งเบิก' || status === 'กำลังจัดสินค้า') {
        pending.push({ name: itemName, qty: reqQty });
      }
    }

    var dateStr = Utilities.formatDate(now, "Asia/Bangkok", "dd MMM yyyy");
    var msg  = "📦 สรุปการจัดสินค้าประจำวัน\n";
    msg     += "วันที่ " + dateStr + " · สรุป ณ 18:00 น.\n\n";
    msg     += "✅ จัดส่งสำเร็จ " + dispatched.length + " รายการ\n";
    msg     += "❌ สินค้าหมด "    + outstock.length   + " รายการ\n";
    msg     += "⏳ รอจัดส่ง "     + pending.length    + " รายการ";

    if (dispatched.length > 0) {
      msg += "\n\n─────────────────\n✅ เบิกสำเร็จวันนี้\n";
      var limit = Math.min(dispatched.length, 20);
      for (var d = 0; d < limit; d++) {
        msg += "• " + dispatched[d].name + " — " + dispatched[d].qty + "\n";
      }
      if (dispatched.length > 20) msg += "... และอีก " + (dispatched.length - 20) + " รายการ\n";
    }

    if (outstock.length > 0) {
      msg += "\n─────────────────\n❌ สินค้าหมด (รอสต๊อกเข้า)\n";
      for (var o = 0; o < outstock.length; o++) {
        msg += "• " + outstock[o].name + " — " + outstock[o].qty + "\n";
      }
    }

    msg += "\n🤖 ส่งอัตโนมัติทุกวัน 18:00 น.";

    var url     = "https://api.line.me/v2/bot/message/push";
    var payload = { "to": LINE_GROUP_ID, "messages": [{ "type": "text", "text": msg }] };
    var options = {
      "method": "post",
      "contentType": "application/json",
      "headers": { "Authorization": "Bearer " + LINE_CHANNEL_TOKEN },
      "payload": JSON.stringify(payload),
      "muteHttpExceptions": true
    };
    var resp = UrlFetchApp.fetch(url, options);
    Logger.log("sendDailyDispatchSummary LINE response: " + resp.getResponseCode());
  } catch (err) {
    Logger.log("sendDailyDispatchSummary error: " + err.toString());
  }
}

// 9. สรุปยอดเติมสินค้าประจำวันและส่งเข้า LINE
// ===============================================
function sendDailyStockSummary() {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName(SURVEY_LOG_SHEET_NAME);
    if (!sheet || sheet.getLastRow() < 2) {
      Logger.log("No survey logs found.");
      return;
    }
    
    var now = new Date();
    var dayStr = Utilities.formatDate(now, "Asia/Bangkok", "dd-MM");
    var yearAD = Utilities.formatDate(now, "Asia/Bangkok", "yyyy");
    var yearBE = String(parseInt(yearAD) + 543);
    var todayStr = dayStr + "-" + yearBE; // แสดงผลเป็นปี พ.ศ.
    
    function isToday(timestamp) {
      if (!timestamp) return false;
      var str = String(timestamp);
      return str.indexOf(dayStr + "-" + yearAD) === 0 || str.indexOf(dayStr + "-" + yearBE) === 0;
    }
    
    var data = sheet.getDataRange().getValues();
    var locationGroups = {};
    var totals = {};
    var hasItems = false;
    var grandTotal = 0;
    
    for (var i = 1; i < data.length; i++) {
      var surveyDate = data[i][0];
      if (!isToday(surveyDate)) continue; // ไม่ใช่ของวันนี้
      
      var floor = data[i][1];      // โลเคชั่น เช่น "1E"
      var name = data[i][2];       // ชื่อสินค้า
      var needToOrder = parseInt(data[i][5]) || 0;
      
      if (needToOrder > 0) {
        hasItems = true;
        
        // จัดกลุ่มตามโลเคชั่น
        if (!locationGroups[floor]) {
          locationGroups[floor] = {};
        }
        if (!locationGroups[floor][name]) {
          locationGroups[floor][name] = 0;
        }
        locationGroups[floor][name] += needToOrder;
        
        // ยอดรวมสินค้า
        if (!totals[name]) {
          totals[name] = 0;
        }
        totals[name] += needToOrder;
        grandTotal += needToOrder;
      }
    }
    
    // จัดรูปแบบข้อความส่ง LINE
    var messageText = "📊 [สรุปยอดจัดเติมสินค้าประจำวัน]\n" +
      "📅 วันที่: " + todayStr + "\n" +
      "━━━━━━━━━━━━━━━━━━\n";
      
    if (hasItems) {
      // 1. แยกตามโลเคชั่น
      for (var loc in locationGroups) {
        messageText += "📍 โลเคชั่น " + loc + "\n";
        for (var name in locationGroups[loc]) {
          var qty = locationGroups[loc][name];
          var unit = getProductUnit(name);
          messageText += "  • " + name + ": " + qty + " " + unit + "\n";
        }
        messageText += "\n";
      }
      
      // 2. ยอดรวมสินค้าทั้งหมดของวันนี้
      messageText += "━━━━━━━━━━━━━━━━━━\n";
      messageText += "📦 ยอดเติมรวมวันนี้: ทั้งหมด " + grandTotal + " ลัง\n";
    } else {
      messageText += "✅ วันนี้เช็คสต๊อกแล้วครบถ้วนดี ไม่พบสินค้าขาดแคลน\n";
    }
    
    // 3. ดึงรายการสินค้าค้างส่ง (สั่งเบิกค้างจากวันก่อนๆ ที่ยังไม่ได้รับการจัดส่ง)
    var allOrders = getInventoryData();
    var pendingGroup = {};
    var pendingCount = 0;
    
    allOrders.forEach(function(item) {
      if (item.status === 'สั่งเบิก' && item.timestamp && !isToday(item.timestamp)) {
        var name = item.itemName;
        var qty = parseInt(item.requestQty) || 0;
        if (qty > 0) {
          if (!pendingGroup[name]) {
            pendingGroup[name] = 0;
          }
          pendingGroup[name] += qty;
          pendingCount++;
        }
      }
    });
    
    if (pendingCount > 0) {
      messageText += "\n⏳ รายการค้างเบิกสะสม (ที่ยังไม่ได้จัดส่ง):\n";
      for (var name in pendingGroup) {
        var qty = pendingGroup[name];
        var unit = getProductUnit(name);
        messageText += "  • " + name + ": " + qty + " " + unit + "\n";
      }
    }
    
    messageText += "━━━━━━━━━━━━━━━━━━\n" +
      "🤖 รายงานอัตโนมัติจากระบบตรวจสต๊อก TRD";
      
    // ส่งไลน์หาห้องแชทกลุ่ม
    var url = "https://api.line.me/v2/bot/message/push";
    var payload = { "to": LINE_GROUP_ID, "messages": [{ "type": "text", "text": messageText }] };
    var options = {
      "method": "post", "contentType": "application/json",
      "headers": { "Authorization": "Bearer " + LINE_CHANNEL_TOKEN },
      "payload": JSON.stringify(payload), "muteHttpExceptions": true
    };
    
    var response = UrlFetchApp.fetch(url, options);
    Logger.log("LINE Daily Summary Response (" + response.getResponseCode() + "): " + response.getContentText());
  } catch (error) {
    Logger.log("Error in sendDailyStockSummary: " + error.toString());
  }
}
